+  const outEdgesOf = new Map<number, LaneEdgeInput[]>();

+    let list = outEdgesOf.get(edge.citingId);
+    if (!list) {

+      outEdgesOf.set(edge.citingId, list);

+  function walk(root: number, node: number, visited: ReadonlySet<number>, hops: number): void {
+    const types = typeOf.get(node) ?? [];

+      const nextIds = new Set((outEdgesOf.get(node) ?? []).map((e) => e.citedId));
+      for (const next of nextIds) {
+        if (visited.has(next)) continue; // cycle guard — this branch is dropped, no destination
+        const nextVisited = new Set(visited);
+        nextVisited.add(next);
+        walk(root, next, nextVisited, hops + 1);
